cmake_minimum_required(VERSION 3.31)
project(grib_consumer LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -DGRIB_PARENT_FLAGS")
if(GRIB_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(grib_cpp SOURCE_DIR "${GRIB_SOURCE_DIR}")
    FetchContent_MakeAvailable(grib_cpp)
    if(GRIB_BUILD_TESTS OR GRIB_BUILD_EXAMPLES OR GRIB_ENABLE_LTO)
        message(FATAL_ERROR "Embedded defaults must be off")
    endif()
    get_directory_property(child_flags DIRECTORY "${GRIB_SOURCE_DIR}" DEFINITION CMAKE_CXX_FLAGS_RELEASE)
    if(NOT child_flags STREQUAL CMAKE_CXX_FLAGS_RELEASE)
        message(FATAL_ERROR "Parent release flags were overwritten")
    endif()
else()
    find_package(grib 0.2.0 EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE grib::grib)
